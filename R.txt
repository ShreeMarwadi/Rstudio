mid=c(34,36,38,40,42,44,46,48,50)
f=c(10,50,99,30,20,14,10,5,2)
h=2
N=sum(f)
#Range(using class boundaries)
range_val=(max(mid)+h/2)-(min(mid)-h/2)
range_val
#Variance and SD
var_val=sum(f*(mid-mean_val)^2)/N
sd_val=sqrt(var_val)
var_val
sd_val
#Quartile Deviation(Q3-Q1)/2
cf=cumsum(f)
get_quartile=function(pos){
qclass=which(cf>=pos)[1]
L=mid[qclass]-h/2
cf_before=ifelse(qclass==1,0,cf[qclass-1])
L+((pos-cf_before)/f[qclass])*h
}
Q1=get_quartile(N/4)
Q3=get_quartile(3*N/4)
QD=(Q3-Q1)/2
Q1;Q3;QD
